function randomGenome(i){

  let s=0;
  for(const tag of ['fire','cold','lightning','poison','dash','cooldown','critical','ichor','temporary','invulnerability','swarm','power','on-hit']){const c=tagCount(g,tag);if(c>1)s+=(c-1)*.05}
  if(W.invulnPower&&A.cooldown)s+=.34;if(W.invulnPower&&F.invulnerability)s+=.24;if(W.perfectDash&&(G.dashDamage||N.coldProc||F.dashRecovery))s+=.28;

  return best;
}

function worldModel(m){
  const g=m.g,H=HEARTS[g.heart],W=WEAPONS[g.weapon],A=ARTIFACTS[g.artifact],F=SOULFRUITS[g.soulfruit];
  const threat=clamp((1-m.hp/m.maxHp)*.48+Math.min(1,m.foes/18)*.34+(+$('hard').value-1)*.06-(g.reflex*.12+g.discipline*.08));
  const profit=clamp(m.ichor/650*.42+(m.chest?.2:0)+g.greed*.2+(H.luck||0)*.18+(A.chestLuck||0)*.15);
  const escape=clamp(g.fear*.34+g.reflex*.22+(A.mobility||0)*.2+(F.mobility||0)*.18-(m.foes>18?.1:0));
  const uncertainty=clamp(.65-m.wave*.055+m.chest*.08+Math.random()*.08);
  const cooldown=clamp(m.cooldown);
  return {threat,profit,escape,uncertainty,cooldown,range:W.reach||1,swarm:clamp(m.foes/25)};
}
