    Chest:m.chest?g.greed*.82+g.luckBias*.3+(A.chestLuck||0)-b.threat*g.fear*.38:-3,
    Explore:g.curiosity*.68+b.uncertainty*.26-b.threat*.25,
    Extract:m.wave>1?b.threat*g.fear*.9+b.profit*g.greed*.72+(m.hp<=1?.65:0):-3
  };
  return (values[action]??0)+syn*.08;
}
function chooseAction(m){
  m.beliefs=worldModel(m);const state=stateKey(m);let total=0;for(const a of ACTIONS)total+=banditEntry(m,state,a).n;
  const scored={};for(const action of ACTIONS){const e=banditEntry(m,state,action);const mean=e.n?e.reward/e.n:0;const explore=Math.sqrt(2*Math.log(total+2)/(e.n+1));scored[action]=priorUtility(m,action)+mean*.32+explore*.26+(Math.random()-.5)*.035}
  m.decisions=scored;const action=Object.entries(scored).sort((a,b)=>b[1]-a[1])[0][0];m.lastState=state;m.lastAction=action;m.lastRewardBase=m.ichor*.018+m.loot*4+m.kills*1.5+m.hp*8;m.action=action;
  if(action==='Extract')m.intention='Preserve the run';else if(action==='Chest')m.intention='Exploit loot opportunity';else if(action==='Dash')m.intention='Create a safe angle';else if(action==='Parry')m.intention='Read the next strike';else if(action==='Power'||action==='Lightning Strike')m.intention='Burst the swarm';else if(action==='Explore')m.intention='Increase map knowledge';else m.intention='Reduce enemy pressure';
  const lines={Attack:`${m.foes} bodies in reach. Keep the rhythm and watch the flanks.`,Heavy:`The opening is narrow. Commit one heavy strike and recover.`,Power:`Cooldown window aligned.   remember(m,lines[action]);
}
